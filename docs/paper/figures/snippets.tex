\begin{figure*}[t]
  \centering
  \input{figures/lemma_overview_tikz}
  \caption{Lemma end to end. The SQL query and a catalog (schema plus an assumption package measured on
    the data and approved by the user) go through the transpiler to a formal spec with a contract for
    \texttt{run\_query}. An LLM agent in a Docker sandbox (\texttt{--network none}, egress only to the
    model API, spec and docs mounted read-only) writes only the body of \texttt{run\_query} and proof
    helpers. The host keeps only those marked regions, rejects \texttt{assume}/\texttt{external\_body}
    and similar, reassembles the program from its own spec, and Verus proves that the body meets the
    contract; only then is it compiled and run. The run uses pinned DuckDB columns, re-checks the catalog
    caps at load, compares output rows with DuckDB and times the query against all-core DuckDB.
    Trusted: the transpiler and spec, a small fixed set of host lemmas and bridges, the admission and
    assembly code, the column-file reader, and the toolchain (Verus, rustc, DuckDB). New trusted code
    must pass the adversary gate.}
  \Description{Four-stage chain: SQL query and catalog, formal spec, sandboxed agent writing Rust, host assembly with Verus proof and compile; below it the checked run and the adversary gate.}
  \label{fig:overview}
\end{figure*}
